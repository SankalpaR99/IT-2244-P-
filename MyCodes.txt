Q1)
Print multiplication table using for loop
1 x 2 = 2
2 x 2 = 4

---ANS--- (mulTable.sh)

echo "Enter multiplier: "
read x

for((i=1;i<=12;i++))
do
ans=$((x*i))
echo "$i x $x = $ans"
done

for((i=1; i<=7; i++))
do
	for((j=1; j<=7-i; j++))
	do
		echo -n " ";
	done
	for((k=1; k<2*i; k++))
	do
		echo -n "*";
	done
	echo ""
done

for((i=6; i>=1; i--))
do
	for((j=1; j<=7-i; j++))
	do
		echo -n " ";
	done
	for((k=1; k<2*i; k++))
	do
		echo -n "*";
	done
	echo ""
done
